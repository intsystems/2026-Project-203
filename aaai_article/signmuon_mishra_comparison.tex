% Comparison with the concurrent SignMuon convergence analysis.
\subsection{\cx{Comparison with the convergence claim for SignMuon}}\label{app:mishra}

\cx{Concurrently, \citet{mishra2026signmuon} introduced the same SignMuon
update, with direction}
$\mathbf D_t=\operatorname{sign}(\operatorname{polar}(\mathbf M_t))/\sqrt{mn}$.
\cx{Absorbing $1/\sqrt{mn}$ into the learning rate gives the step of
Section~\ref{sec:theory}. Their abstract attributes an
$\mathcal O(T^{-1/2})$ stationarity rate to SignMuon; the rate theorem is
stated for a gradient-sign instantiation.}

\subsubsection*{\cx{Sign-error decomposition}}
\cx{For a transmitted sign matrix $\mathbf S_t$ and
$\mathbf D_t=\mathbf S_t/\sqrt{mn}$, define}
\[
\begin{aligned}
\mathcal G_T&=\frac1T\sum_{t=0}^{T-1}\mathbb E\!\left[
\frac{\|\nabla f(\mathbf X_t)\|_1}{\sqrt{mn}}\right],\\
q_{ij,t}&=\Pr\!\left(\mathbf S_{t,ij}\neq
\operatorname{sign}([\nabla f(\mathbf X_t)]_{ij})\mid\mathbf X_t\right).
\end{aligned}
\]
\cx{The identity}
\begin{equation}\label{eq:mishra_identity}
\mathbb{E}\bigl[\langle \nabla f(\mathbf{X}_t),\,\mathbf{D}_t\rangle \,\big|\, \mathbf{X}_t\bigr]
=\frac{1}{\sqrt{mn}}\sum_{i,j}\bigl|[\nabla f(\mathbf{X}_t)]_{ij}\bigr|\,\bigl(1-2q_{ij,t}\bigr)
\end{equation}
\cx{and the spectral-smoothness descent lemma give, for any sign oracle,}
\begin{equation}\label{eq:mishra_generic}
\mathcal{G}_T
\;\le\;
\frac{f(\mathbf{X}_0)-f^\ast}{\eta T}
+\frac{L_\ast}{2}\,\eta
+\underbrace{\frac{2}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\left[\frac{1}{\sqrt{mn}}\sum_{i,j}\bigl|[\nabla f(\mathbf{X}_t)]_{ij}\bigr|\,q_{ij,t}\right]}_{=:\,R_T}.
\end{equation}
\cx{A convergence rate therefore requires control of $R_T$.}

\subsubsection*{\cx{Gradient-sign convergence rate}}
\cx{The residual estimate uses
$\mathbf S_t=\operatorname{sign}(\widetilde{\mathbf G}_t)$, where
$\widetilde{\mathbf G}_t$ is an unbiased stochastic gradient with
coordinatewise variance at most $\sigma_{ij}^2/n_b$ at batch size $n_b$.
A Markov--Jensen argument gives}
\[
\bigl|[\nabla f(\mathbf X_t)]_{ij}\bigr|q_{ij,t}
\le\frac{\sigma_{ij}}{\sqrt{n_b}},\qquad
R_T\le\frac{2\|\sigma\|_1}{\sqrt{mn\,n_b}},\qquad
\|\sigma\|_1=\sum_{i,j}\sigma_{ij}.
\]
\cx{With $n_b=T$ and $\eta\asymp T^{-1/2}$,
\eqref{eq:mishra_generic} yields $\mathcal G_T=\mathcal O(T^{-1/2})$.
This update is normalized SignSGD without momentum, analysed in single-worker
and majority-vote forms under spectral smoothness
\citep{bernstein2018signsgd,mishra2026signmuon}. For SignMuon, the polar
transformation introduces additional sign errors that can persist with exact
gradients. Their analysis leaves the corresponding $R_T$ unestimated.}

\subsubsection*{\cx{Evaluation on the SignMuon counterexample}}
\cx{For the $4\times4$ gradient $\mathbf G$ of Theorem~\ref{th:1},
Proposition~\ref{prop:reduction} gives the constant sign matrix
$\mathbf S=\operatorname{sign}(\operatorname{polar}(\mathbf G))$
for every momentum coefficient and either momentum rule. Thus $q_{ij}$ is
the indicator of a sign mismatch, and \eqref{eq:mishra_identity} becomes}
\begin{equation}\label{eq:mishra_instance}
\sum_{i,j}|\mathbf{G}_{ij}|\,(1-2q_{ij})
=\bigl\langle \mathbf{G},\,\operatorname{sign}(\operatorname{polar}(\mathbf{G}))\bigr\rangle
=-\tfrac{42468}{103}\;<\;0 .
\end{equation}
\cx{Since $\sqrt{mn}=4$,}
\begin{equation}\label{eq:mishra_dominates}
R_T=\mathcal{G}_T+\tfrac{10617}{103}
\qquad\text{for every } T .
\end{equation}
\cx{The normalized SignMuon step therefore satisfies}
\[
f(\mathbf X_t)-f(\mathbf X_{t-1})
=-\eta\langle\mathbf G,\mathbf D_t\rangle
=\eta\,\tfrac{10617}{103}>0.
\]
\cx{The smooth bounded-below modification of Remark~\ref{rem:efsm_bdd}
preserves this trajectory and all the identities above. On that objective,
\eqref{eq:mishra_generic} holds with a finite $f^\ast$, yet
$R_T>\mathcal G_T$ at every horizon. Thus the generic inequality is compatible
with ascent, while the proved rate applies to the gradient-sign update.}
