\subsection{Nesterov momentum}\label{app:nesterov_conv}

We prove convergence for Nesterov momentum in
Algorithm~\ref{alg:fed_serverlmo}. With
$\beta=1-\mu$ and $a=1-\beta$:
\begin{align}
 M_{t,i}^{(j)}&=aM_{t-1,i}^{(j)}+\beta G_{t,i}^{(j)},\nonumber\\
 V_{t,i}^{(j)}&=aM_{t,i}^{(j)}+\beta G_{t,i}^{(j)},\label{eq:nest_momentum}\\
 g_{t,i}^{(j)}&=g_{t-1,i}^{(j)}+
 C_i^\uparrow(V_{t,i}^{(j)}-g_{t-1,i}^{(j)}).\nonumber
\end{align}
Initialize $M_{0,i}^{(j)}=g_{0,i}^{(j)}=0$, $W_0=X_0$, and set
$M_{-1,i}^{(j)}=V_{0,i}^{(j)}=0$ for every layer $i$ and worker $j$.
The oracle is the exact measurable polar
selection, with zero direction at zero. Put $\rho_i=\sqrt{r_i}$ and
$\alpha_D=1/d_{\max}$, and take $\alpha_P=1$ for the exact downlink and
$\alpha_P=\alpha_D$ for the compressed downlink.
Let $\sigma_i^2$ bound each worker's Frobenius noise variance in block $i$;
Assumption~\ref{as:3} permits $\sigma_i=\sigma$.

\begin{theorem}[\textbf{Nesterov convergence}]\label{thm:nesterov_conv}
With the initialization above, the Nesterov updates
\eqref{eq:nest_momentum} have the following guarantees.
\begin{itemize}
\item[\emph{(i)}] Under Assumptions~\ref{as:1}--\ref{as:3} with ordinary
smoothness, EF21-MuonUSign and EF21-MuonSign satisfy
\[
 \frac1T\sum_{t<T}\sum_i\mathbb E\|\nabla_i f(X_t)\|_*^2
 =\mathcal O(T^{-1/2})
\]
with $\beta=T^{-1/2}$ and sharp steps
$\eta_{t,i}=\gamma\|g_{t,i}\|_*$, $\gamma=c\beta$.
An explicit sufficient choice of the fixed $c>0$ is given in the proof of
Lemma~\ref{lem:nest_smooth_tracking}.
\item[\emph{(ii)}] Under the generalized form of Assumption~\ref{as:2} and
Assumptions~\ref{as:1},~\ref{as:3}, EF21-MuonUSign with exact downlink,
$S=T+2$, $\beta=S^{-1/2}$ and learning rates
$\eta_{t,i}=\eta_i S^{-3/4}$ satisfies
\[
 \min_{0\le t\le T}\sum_i\frac{\eta_i}{\bar\eta}
       \mathbb E\|\nabla_i f(X_t)\|_*
 =\mathcal O(T^{-1/4}),\qquad
 \bar\eta=\frac1p\sum_i\eta_i,
\]
for any fixed $\eta_i>0$.
\end{itemize}
All constants are independent of $T$.
\end{theorem}

The proof has two parts: a squared-error tracking estimate for ordinary
smoothness and a first-moment estimate for generalized smoothness.
For both, cut a round just before the server step. A state consists of
$(X,W,M,M_{\rm prev},g)$, and its successor first forms $(X',W')$ by the
step and downlink, then queries fresh gradients at $W'$, and finally applies
\eqref{eq:nest_momentum}. Denote conditional expectation given this state by
$\mathbb E_t$. This is the same rotation as in
Proposition~\ref{prop:instance}: the initial step is zero and
$X_{t+1}^{\rm rot}=X_t^{\rm paper}$. The previous buffer is carried as an
extra analysis state. In particular,
\begin{align}
 V_i^{(j)}&=(2-\beta)M_i^{(j)}-aM_{{\rm prev},i}^{(j)},\nonumber\\
 V_i^{(j)\prime}-V_i^{(j)}
 &=(2-\beta)(M_i^{(j)\prime}-M_i^{(j)})
       -a(M_i^{(j)}-M_{{\rm prev},i}^{(j)}).\label{eq:nest_increment}
\end{align}
Thus tracking the Nesterov target also requires controlling momentum
increments. Write $\langle z_j\rangle_j=N^{-1}\sum_jz_j$.

\begin{lemma}[\textbf{Smooth Nesterov tracking}]\label{lem:nest_smooth_tracking}
Under Assumptions~\ref{as:1}--\ref{as:3} with ordinary smoothness and the
initialization above, there is a
constant $c_0>0$ such that, for $0<c\le c_0$, $0<\beta\le1$ and
$\gamma=c\beta$, a nonnegative potential $\Psi$ satisfies
\[
 \mathbb E_t\Psi'+\frac\gamma4\sum_i\|\nabla_i f(X)\|_*^2
 \le\Psi+\gamma\beta\mathcal S,\qquad \Psi_0\le\mathcal I.
\]
The constants $c_0,\mathcal I,\mathcal S$ are independent of $\beta,T$;
explicit values are given in the proof.
\end{lemma}
\begin{paperproof}
We track the averaged momentum error, the local momentum errors, the
uplink errors, the momentum increments, and the downlink gap. For each
layer, put $\ell_i^2=\tilde L_i^2$ and define
\begin{align*}
 E_i&=\|\langle M_i^{(j)}\rangle_j-\nabla_i f(X)\|_F^2,\\
 F_i&=\langle\|M_i^{(j)}-\nabla_i f_j(X)\|_F^2\rangle_j,\\
 H_i&=\langle\|V_i^{(j)}-g_i^{(j)}\|_F^2\rangle_j,\\
 K_i&=\|\langle M_i^{(j)}-M_{{\rm prev},i}^{(j)}\rangle_j\|_F^2,\\
 J_i&=\langle\|M_i^{(j)}-M_{{\rm prev},i}^{(j)}\|_F^2\rangle_j,
 \qquad P_i=\|W_i-X_i\|_F^2.
\end{align*}
Let $A_i=\langle\|\nabla_i f_j(X')-\nabla_i f_j(X)\|_F^2\rangle_j$
and $B_i=\langle\|\nabla_i f_j(W')-\nabla_i f_j(X')\|_F^2\rangle_j$.
Conditional centering, Young's inequality, and compressor contractivity give
\begin{align}
 \mathbb E_t F_i'&\le(1-\beta/2)F_i+2A_i/\beta\nonumber\\
 &\quad+\beta^2(1+2/\beta)B_i+\beta^2\sigma_i^2,\nonumber\\
 \mathbb E_t J_i'&\le3\beta^2(F_i+A_i+B_i)+\beta^2\sigma_i^2,
 \label{eq:nest_tracking}\\
 \mathbb E_t H_i'&\le(1-\alpha_D/2)H_i
       +\frac{16}{\alpha_D}\mathbb E_tJ_i'
       +\frac4{\alpha_D}J_i.\nonumber
\end{align}
The same bounds hold for $E_i',K_i'$ with $F_i,J_i$ replaced by
$E_i,K_i$ and $\sigma_i^2$ replaced by $\sigma_i^2/N$.
Indeed, average the centered worker noises before squaring; their conditional
cross terms vanish. Jensen bounds the averaged gradient movements by
$A_i,B_i$. Thus worker averaging is retained in the direction error, while
the local recurrences still control compression.
For the last inequality, \eqref{eq:nest_increment} yields, for each worker,
$\|V_i^{(j)\prime}-V_i^{(j)}\|_F^2
 \le8\|M_i^{(j)\prime}-M_i^{(j)}\|_F^2
       +2\|M_i^{(j)}-M_{{\rm prev},i}^{(j)}\|_F^2$;
apply the contracted Young inequality to
$(V_i^{(j)}-g_i^{(j)})+(V_i^{(j)\prime}-V_i^{(j)})$
before averaging. Smoothness gives
$A_i\le\ell_i^2\gamma^2\|g_i\|_*^2$ and
$B_i\le\ell_i^2P_i'$, where $g_i=\langle g_i^{(j)}\rangle_j$.
The downlink satisfies
\[
 P_i'\le(1-\alpha_P/2)P_i+
       \frac{2\tau_i}{\alpha_P}\gamma^2\|g_i\|_*^2,
\]
with $\tau_i=0$ for the exact downlink and $\tau_i=r_i$ for the
scaled-sign downlink.

To combine these recurrences with objective descent, choose the following
nonnegative weights:
\begin{align*}
 b_i&=8\rho_i^2,& h_i&=2b_i/\alpha_D,\\
 j_i&=b_i+4h_i/\alpha_D,& R_i&=16h_i/\alpha_D+j_i,\\
 \widehat R_i&=R_i-b_i=40b_i/\alpha_D^2,
 &u_i&=2b_i/\beta+6\beta R_i,\\
 e_i&=2b_i/\beta+6\beta b_i,
 &v_i&=6\beta\widehat R_i,\\
 I_i&=[u_i\beta^2(1+2/\beta)+3\beta^2R_i]\ell_i^2,
 &p_i&=2I_i/\alpha_P,\\
 \zeta_i^{\rm N}&=2u_i\ell_i^2/\beta+3\beta^2R_i\ell_i^2
                   +(I_i+p_i)2\tau_i/\alpha_P.
\end{align*}
The identities $e_i+v_i=u_i$ and
$b_i+\widehat R_i=R_i$ preserve the movement and broadcast coefficients.
Moreover, $e_i\beta/2-3\beta^2b_i=b_i$,
$v_i\beta/2-3\beta^2\widehat R_i=0$,
$h_i\alpha_D/2=b_i$, and $j_i-b_i=4h_i/\alpha_D$
absorb the global and local tracking errors.
Indeed, Jensen's inequality and \eqref{eq:nest_increment} imply
$\|g_i-\nabla_i f(X)\|_F^2\le3(E_i+H_i+K_i)$, and the polar
identities imply
\begin{align*}
 \langle\nabla_i f(X),X_i'-X_i\rangle
 &\le\gamma b_i(E_i+H_i+K_i)\\
 &\quad-\tfrac\gamma4\|\nabla_i f(X)\|_*^2
          -\tfrac\gamma4\|g_i\|_*^2.
\end{align*}
Consequently the nonnegative potential
\[
 \Psi=f(X)-f_*+\sum_i\gamma\bigl[
 e_iE_i+v_iF_i+h_iH_i+b_iK_i+(j_i-b_i)J_i+p_iP_i\bigr]
\]
satisfies, whenever $2L_i\gamma+4\zeta_i^{\rm N}\gamma^2\le1$,
\begin{equation}\label{eq:nest_smooth_drift}
 \mathbb E_t\Psi'+\frac\gamma4\sum_i\|\nabla_i f(X)\|_*^2
 \le\Psi+\gamma\sum_i\nu_i\sigma_i^2,
\end{equation}
where $\nu_i=\beta^2[(e_i+b_i)/N+v_i+\widehat R_i]$.
Here smooth descent contributes $L_i\gamma^2\|g_i\|_*^2/2$;
all tracking movement terms contribute at most
$\zeta_i^{\rm N}\gamma^3\|g_i\|_*^2$, which the stated condition absorbs.

It remains to choose a sharp-step scale that satisfies this condition
uniformly over $\beta$. Define
\begin{align*}
 Z_i^{\rm N}={}&(4b_i+15R_i)\ell_i^2\\
 &+(6b_i+21R_i)\ell_i^2(1+2/\alpha_P)\,2\tau_i/\alpha_P.
\end{align*}
Direct substitution gives
$\beta^2\zeta_i^{\rm N}\le Z_i^{\rm N}$,
$\beta u_i\le2b_i+6R_i$, and
$\nu_i\le\beta(9b_i/N+7\widehat R_i)$.
Take
$c_0=[1+\sum_i(2L_i+2\sqrt{Z_i^{\rm N}})]^{-1}$.
Every $0<c\le c_0$ satisfies
$c(2L_i+2\sqrt{Z_i^{\rm N}})\le1$ for every layer.
Then $\gamma=c\beta$ ensures the condition in
\eqref{eq:nest_smooth_drift}. Put
\begin{align*}
 \mathcal I&=f(X_0)-f_*+
 c\sum_i\bigl[8b_i\|\nabla_i f(X_0)\|_F^2
 +6\widehat R_i\langle\|\nabla_i f_j(X_0)\|_F^2\rangle_j\bigr],\\
 \mathcal S&=\sum_i(9b_i/N+7\widehat R_i)\sigma_i^2.
\end{align*}
Zero initialization gives $\Psi_0\le\mathcal I$, and
$\sum_i\nu_i\sigma_i^2\le\beta\mathcal S$ proves the drift claimed
in the lemma.
\end{paperproof}

\begin{lemma}[\textbf{Generalized-smooth Nesterov tracking}]\label{lem:nest_general_tracking}
Under Assumptions~\ref{as:1},~\ref{as:3}, generalized smoothness, and the
initialization above, use the exact downlink. Fix
$\eta_i>0$ and set $s_i=\eta_iS^{-3/4}$, $\beta=S^{-1/2}$ for $S\ge2$.
There are a nonnegative potential $\Phi$ and finite nonnegative constants
$K,B,I$, independent of $S$, such that
\[
 \mathbb E_t\Phi'+\sum_i s_i\|\nabla_i f(X)\|_*
 \le(1+K/S)\Phi+B/S,\qquad \Phi_0\le I.
\]
\end{lemma}
\begin{paperproof}
The exact downlink gives $W=X$. We first separate the averaged gradient
noise from the tracking bias, and then bound the gradient-dependent
smoothness terms by the objective gap. Let $q=\sqrt{1-\alpha_D}<1$ and let
$Z_i'=aZ_i+\beta\langle\varepsilon_i^{(j)}\rangle_j$, $Z_{0,i}=0$,
be an analysis-only noise EMA. Replace $F_i,H_i,J_i$ by their first-moment
versions $F_i^{(1)},H_i^{(1)},J_i^{(1)}$ (remove the squares), and put
$D_i=\|\langle M_i^{(j)}\rangle_j-\nabla_i f(X)-Z_i\|_F$.
For $d_i^g=\|\nabla_i f(X')-\nabla_i f(X)\|_F$ and
$d_i^\ell=\langle\|\nabla_i f_j(X')-\nabla_i f_j(X)\|_F\rangle_j$,
the same updates give
\begin{align*}
 \mathbb E_tD_i'&\le a(D_i+d_i^g),\\
 \mathbb E_tF_i^{(1)\prime}&\le a(F_i^{(1)}+d_i^\ell)+\beta\sigma_i,\\
 \mathbb E_tJ_i^{(1)\prime}&\le\beta(F_i^{(1)}+d_i^\ell+\sigma_i),\\
 \mathbb E_tH_i^{(1)\prime}&\le qH_i^{(1)}+
                 2q\mathbb E_tJ_i^{(1)\prime}+qJ_i^{(1)},\\
 \mathbb E_t\|Z_i'\|_F^2&\le a\|Z_i\|_F^2+\beta^2\sigma_i^2/N.
\end{align*}
The last line uses independent centered worker noise. The first line is
noise-free after debiasing. Define
\begin{align*}
 \delta_i&=2s_i\rho_i/\beta,& h_i^g&=4s_i\rho_i/(1-q),\\
 j_i^g&=2s_i\rho_i+qh_i^g,& R_i^g&=2qh_i^g+j_i^g,\\
 z_i^g&=s_i\rho_i/(\beta\sqrt\beta).
\end{align*}
Since $\|g_i-\nabla_i f(X)\|_F\le
D_i+H_i^{(1)}+J_i^{(1)}+\|Z_i\|_F$, these recursions and
$2\|Z_i\|_F\le\|Z_i\|_F^2/\sqrt\beta+\sqrt\beta$ show that
\[
 \Phi=f(X)-\bar f_*+\sum_i\bigl[
 \delta_iD_i+R_i^gF_i^{(1)}+h_i^gH_i^{(1)}
                  +j_i^gJ_i^{(1)}+z_i^g\|Z_i\|_F^2\bigr]
\]
obeys
\begin{align}
 \mathbb E_t\Phi'+\sum_i s_i\|\nabla_i f(X)\|_*
 &\le\Phi+\sum_i\left[A_i^g C_i^g(X)+A_i^\ell C_i^\ell(X)+n_i\right],
 \label{eq:nest_general_drift}\\
 A_i^g&=s_i^2/2+\delta_i a s_i,\quad A_i^\ell=R_i^g s_i,\nonumber\\
 n_i&=2R_i^g\beta\sigma_i+
                 z_i^g\beta^2\sigma_i^2/N+s_i\rho_i\sqrt\beta,\nonumber
\end{align}
where $\bar f_*=N^{-1}\sum_j f_j^*$,
$C_i^g(X)=L_i^0+L_i^1\|\nabla_i f(X)\|_*$, and
$C_i^\ell(X)=\langle L_{i,j}^0+L_{i,j}^1\|\nabla_i f_j(X)\|_*\rangle_j$.
Indeed, generalized smoothness bounds the gradient movements by
$d_i^g\le s_i C_i^g(X)$, $d_i^\ell\le s_i C_i^\ell(X)$.
Its descent inequality contributes $s_i^2C_i^g(X)/2$ and
$2s_i\rho_i\|g_i-\nabla_i f(X)\|_F$; the weighted recursions absorb
the latter term, giving \eqref{eq:nest_general_drift}.

To bound the right-hand side of \eqref{eq:nest_general_drift}, we use the
lower bounds on the local objectives. They give finite nonnegative
constants $a_i^g,b_i^g,a_i^\ell,b^\ell$ such that
\[
 C_i^g(X)\le a_i^g+b_i^g(f(X)-\bar f_*),\quad
 C_i^\ell(X)\le a_i^\ell+b^\ell(f(X)-\bar f_*).
\]
For completeness, set $M=1+\sum_iL_i^1+\sum_{j,i}L_{i,j}^1$,
$o=\sum_i L_i^0/M$ and $\tilde o=\langle\sum_i L_{i,j}^0/M\rangle_j$.
A generalized-smooth descent step of radius $1/M$ in each block gives
$\sum_i\|\nabla_i f(X)\|_*\le2M(f(X)-\bar f_*)+o$,
and the analogous bound holds for each $f_j-f_j^*$.
Thus one can take
$a_i^g=L_i^0+L_i^1o$, $b_i^g=2ML_i^1$,
$a_i^\ell=\langle L_{i,j}^0\rangle_j+M\tilde o$ and $b^\ell=2M^2$.
For the chosen $s_i$ and $\beta$, each remaining coefficient is of order
$1/S$. More explicitly, define
\begin{align*}
 c_i^g&=\tfrac12+2\rho_i,&
 c_i^\ell&=2\rho_i+12\rho_i q/(1-q),\\
 c_i^n&=2c_i^\ell\sigma_i+\rho_i\sigma_i^2/N+\rho_i.
\end{align*}
The coefficients above satisfy
$A_i^g\le(s_i^2/\beta)c_i^g$,
$A_i^\ell\le(s_i^2/\beta)c_i^\ell$ and
$n_i\le s_i\sqrt\beta\,c_i^n$.
Both
$s_i^2/\beta=\eta_i^2/S$ and $s_i\sqrt\beta=\eta_i/S$.
Consequently \eqref{eq:nest_general_drift} implies
\[
 \mathbb E_t\Phi'+\sum_i s_i\|\nabla_i f(X)\|_*
 \le(1+K/S)\Phi+B/S,
\]
where
\begin{align*}
 K&=\sum_i\eta_i^2(c_i^g b_i^g+c_i^\ell b^\ell),\\
 B&=\sum_i[\eta_i^2(c_i^g a_i^g+c_i^\ell a_i^\ell)+\eta_i c_i^n],\\
 I&=f(X_0)-\bar f_*+\sum_i\bigl[2\eta_i\rho_i\|\nabla_i f(X_0)\|_F\\
 &\hspace{38mm}+\eta_i c_i^\ell
                  \langle\|\nabla_i f_j(X_0)\|_F\rangle_j\bigr].
\end{align*}
Initialization gives $\Phi_0\le I$. This proves the lemma.
\end{paperproof}

\begin{paperproof}[{Proof of Theorem~\ref{thm:nesterov_conv}}]
For part~(i), sum the
inequality of Lemma~\ref{lem:nest_smooth_tracking} over $T+1$ rotated
rounds. Discard the nonnegative final potential and the duplicated initial
gradient term. With $\beta=T^{-1/2}$ and $(T+1)/T\le2$, this gives
\begin{equation}\label{eq:nest_smooth_rate}
 \frac1T\sum_{t<T}\sum_i\mathbb E\|\nabla_i f(X_t)\|_*^2
 \le\frac{4\mathcal I/c+8\mathcal S}{\sqrt T}.
\end{equation}

For part~(ii), apply Lemma~\ref{lem:nest_general_tracking} with $S=T+2$.
Divide its inequality at round $t$ by $(1+K/S)^{t+1}$ and sum.
For any $k\le S$, the total expected gradient cost is at most
$e^K(I+B)$, since $(1+K/S)^k\le e^K$.
Taking $k=T+1$, dividing by
$\bar\eta(T+1)S^{-3/4}$, and using $S/(T+1)\le2$ gives
$2e^K(I+B)/(\bar\eta S^{1/4})$ for the rotated average.
Its minimum of expectations is no larger. The round rotation transfers
this minimum to the paper's iterates $0,\ldots,T$: it removes a duplicated
initial model and adds the final paper model. This proves the second claim.
All conditional recursions are integrable: the smooth proof propagates second
moments, and the generalized proof propagates the displayed first moments and
the squared noise EMA. Finite-product integration justifies these iterations.
\end{paperproof}
